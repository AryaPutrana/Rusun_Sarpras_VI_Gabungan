function previewFile() {
    const input = document.getElementById('foto');
    const fileInfo = document.getElementById('file-info');
    const fileName = document.getElementById('file-name');
    const fileSize = document.getElementById('file-size');
    
    if (input.files && input.files[0]) {
        const file = input.files[0];
        fileName.textContent = file.name;
        fileSize.textContent = formatFileSize(file.size);
        fileInfo.style.display = 'block';

            name: file.name,
            size: file.size,
            type: file.type
        });
    } else {
        fileInfo.style.display = 'none';
    }
}

function formatFileSize(bytes) {
    if (bytes < 1024) return bytes + ' B';
    if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(2) + ' KB';
    return (bytes / (1024 * 1024)).toFixed(2) + ' MB';
}
</script>

SKENARIO DEBUGGING BERDASARKAN LOG
═══════════════════════════════════════════════════════════════════════

SKENARIO A: Log "has_file" = false
────────────────────────────────────────────────────────────────────
Artinya: Form tidak mengirim file ke server

Penyebab kemungkinan:
  1. Form tidak ada enctype="multipart/form-data"
  2. Input name bukan "foto"
  3. JavaScript menghapus file sebelum submit
  4. Browser bug/issue

Solusi:
  → Cek form enctype di HTML source
  → Cek Network tab Form Data
  → Test di browser berbeda


SKENARIO B: Log "has_file" = true, tapi "storeUploaded returned null"
────────────────────────────────────────────────────────────────────
Artinya: File sampai ke server tapi gagal disimpan

Penyebab kemungkinan:
  1. Storage permission issue
  2. Disk penuh
  3. Exception di PhotoStorage::storeUploaded()

Solusi:
  → Cek storage/logs/laravel.log untuk exception
  → Cek disk space: df -h
  → Cek permission: ls -la storage/app/evidence/monitoring


SKENARIO C: Log "Photo uploaded successfully" tapi foto tidak berubah
────────────────────────────────────────────────────────────────────
Artinya: File tersimpan tapi database tidak update atau cache browser

Penyebab kemungkinan:
  1. Database update gagal (tapi tidak ada exception)
  2. Field 'foto' tidak di-update (tidak masuk $data)
  3. Browser cache

Solusi:
  → Cek database: SELECT foto, foto_disk, updated_at FROM water_monitorings WHERE id=X
  → Jika foto berubah di DB tapi tampilan sama → Hard refresh (Ctrl+F5)
  → Jika foto tidak berubah di DB → Cek log "Monitoring updated successfully"


SKENARIO D: Tidak ada log sama sekali
────────────────────────────────────────────────────────────────────
Artinya: Request tidak sampai ke controller update()

Penyebab kemungkinan:
  1. Form action salah
  2. Route tidak match
  3. Middleware reject request
  4. Form submit tidak terjadi

Solusi:
  → Cek form action di HTML source
  → Cek browser Network tab - ada request ke /monitoring/{id}?
  → Test submit form manual (tanpa JavaScript)

AFTER:
```php
if ($request->hasFile('foto')) {
    $file = $request->file('foto');
    
    // Explicit validation
    if (!$file->isValid()) {
        return back()->withInput()->with('error', 'File upload gagal. Coba lagi.');
    }
    
    // Try upload with more error info
    try {
        $path = $file->store(PhotoStorage::DIR_MONITORING, PhotoStorage::EVIDENCE);
        
        if (!$path) {
            \Log::error('File store() returned false');
            return back()->withInput()->with('error', 'Gagal menyimpan file ke storage.');
        }
        
        $data['foto'] = $path;
        $data['foto_disk'] = PhotoStorage::EVIDENCE;
        
        \Log::info('Photo stored', ['path' => $path]);
        
    } catch (\Exception $e) {
        \Log::error('Photo store exception', ['error' => $e->getMessage()]);
        return back()->withInput()->with('error', 'Error: ' . $e->getMessage());
    }
}
```
